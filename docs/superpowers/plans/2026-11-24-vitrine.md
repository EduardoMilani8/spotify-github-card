# Vitrine de álbuns — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Um endpoint `/api/showcase` que lê 3 álbuns de uma playlist do Spotify e devolve um SVG com o carrossel aprovado, para ficar ao lado do card do vinil no perfil.

**Architecture:** Mesmo padrão do card existente: funções puras (seleção de álbuns, renderização) mais um orquestrador com dependências injetadas (`env`, `fetchFn`, `agora`, `log`) e fallback que sempre devolve um SVG válido. O código novo fica em `src/vitrine/` e reaproveita `obterAccessToken`, `baixarCapaBase64`, `escolherImagemCapa`, `ajustarTexto` e `escaparXml`.

**Tech Stack:** TypeScript ~5.9 (ESM, NodeNext, imports com `.js`), Vitest, fast-xml-parser (só nos testes), Vercel Functions (Node 22).

**Spec:** `docs/superpowers/specs/2026-11-24-vitrine-design.md`

## Global Constraints

- TypeScript fica em `~5.9` e Node em `22.x`; nenhuma dependência nova.
- Imports relativos sempre com extensão `.js` (NodeNext).
- O endpoint responde **sempre 200** com `Content-Type: image/svg+xml; charset=utf-8`.
- Cabeçalhos da vitrine: `Cache-Control: no-cache, no-store, max-age=0` e `CDN-Cache-Control: max-age=3600, stale-while-revalidate=86400`.
- Fundo `#0e4429`; verdes `#9be9a8`, `#39d353`, `#006d32`; viewBox 400×150.
- Todo texto vindo do Spotify passa por `escaparXml`.
- Nomes longos nunca são cortados: a fonte diminui de 11 até 9 e depois o texto rola.
- Idioma padrão `en`; `pt` via `?lang=pt`.
- Variável de ambiente: `SPOTIFY_SHOWCASE_PLAYLIST` (link ou ID da playlist).

## Review Focus

1. **Álbum sem imagens** (`images: []`): `capaUrl` deve ser `null` e a vitrine usa a capa verde de fallback. Teste na Task 3.
2. **Nome com CJK ou emoji longo**: o SVG continua XML válido e o título rola. Teste na Task 4.
3. **Virada de mês**: o cabeçalho usa o mês em UTC. Teste na Task 4.
4. **Resposta da playlist sem `items` ou com entradas malformadas**: vira vitrine vazia, nunca erro 500. Teste na Task 5.
5. **Capa que falha ao baixar**: o carrossel continua com 3 álbuns e capa de fallback. Teste na Task 5.

import { getApi, type FetchFn } from '../spotify.js'
import type { ItensPlaylist } from './tipos.js'

const ID_SPOTIFY = /^[A-Za-z0-9]{22}$/
const CAMPOS = 'items(item(type,is_local,album(id,name,artists(name),images)))'

/** Aceita o link da playlist, a URI `spotify:playlist:<id>` ou só o ID. */
export function lerIdPlaylist(valor: string | null | undefined): string | null {
  const texto = valor?.trim() ?? ''
  const achado = texto.match(/playlist[/:]([A-Za-z0-9]{22})/)
  const id = achado ? achado[1] : texto
  return ID_SPOTIFY.test(id) ? id : null
}

export async function buscarItensPlaylist(token: string, id: string, fetchFn: FetchFn = fetch): Promise<ItensPlaylist> {
  const query = new URLSearchParams({ limit: '20', fields: CAMPOS })
  const resposta = await getApi(`/playlists/${id}/items?${query}`, token, fetchFn)
  return (await resposta.json()) as ItensPlaylist
}
```

export function escolherAlbuns(itens: ItemPlaylist[]): Album[] {
  const albuns: Album[] = []
  const vistos = new Set<string>()
  for (const { item } of itens) {
    const album = item?.album
    if (item?.type !== 'track' || item.is_local || !album?.id || vistos.has(album.id)) continue
    vistos.add(album.id)
    albuns.push(paraAlbum({ ...album, id: album.id }))
    if (albuns.length === MAX_ALBUNS) break
  }
  return albuns
}
```

/** Imagem da capa (ou fallback verde) definida uma vez; capa, reflexo e selo usam <use>. */
function defImagem(i: number, capa: string | null): string {
  const conteudo = capa
    ? `<image href="${capa}" x="-42" y="-42" width="84" height="84" preserveAspectRatio="xMidYMid slice"/>`
    : '<rect x="-42" y="-42" width="84" height="84" fill="#006d32"/><circle r="14" fill="#39d353"/>'
  return `<g id="img-${i}">${conteudo}</g>`
}

function vinil(i: number, girando: boolean): string {
  const selo = i >= 0 ? `<g transform="scale(.2857)"><use href="#img-${i}" clip-path="url(#selo)"/></g>` : '<circle r="12" fill="#39d353"/>'

const capaComReflexo = (i: number): string =>
  `<g mask="url(#reflexo)"><g transform="translate(0 86) scale(1 -1)"><use href="#img-${i}" clip-path="url(#recorte-capa)"/></g></g>` +

/** Título "Álbum · Artistas" centrado em x; encolhe de 11 a 9px e depois rola. */
function titulo(i: number, album: Album, x: number, larguraMax: number, sincronizado: boolean): Parte {
  const id = `v${i}`
  const aj = ajustarTexto(`${album.nome} · ${album.artistas}`, larguraMax, 11, 9)
  const conteudo = `${escaparXml(album.nome)} <tspan fill-opacity=".65">· ${escaparXml(album.artistas)}</tspan>`
  const atributos = `y="138" font-size="${aj.fontSize}" fill="#ffffff" font-family="${FONTE}"`
  if (aj.rolagem === 0) {
    return { css: '', defs: '', elementos: `<text x="${x}" text-anchor="middle" ${atributos}>${conteudo}</text>` }
  }
  const esquerda = x - larguraMax / 2
  // No carrossel o título só aparece ~3s: a rolagem acompanha o ciclo de 12s do álbum.
  const css = sincronizado
    ? `@keyframes rolar-${id}{0%,6%{transform:translateX(0)}20%,100%{transform:translateX(-${aj.rolagem}px)}}` +
      `.rolar-${id}{animation:rolar-${id} 12s ease-in-out infinite;animation-delay:var(--d)}`
    : `@keyframes rolar-${id}{0%,15%{transform:translateX(0)}70%,85%{transform:translateX(-${aj.rolagem}px)}100%{transform:translateX(0)}}` +
      `.rolar-${id}{animation:rolar-${id} 9s ease-in-out infinite}`
  return {
    css,
    defs: `<clipPath id="recorte-${id}"><rect x="${esquerda}" y="${138 - aj.fontSize * 1.2}" width="${larguraMax}" height="${aj.fontSize * 1.6}"/></clipPath>`,
    elementos: `<g clip-path="url(#recorte-${id})"><text class="rolar-${id}" x="${esquerda}" ${atributos}>${conteudo}</text></g>`,
  }
}

function carrossel(albuns: Album[]): Parte {
  const discos = albuns.map((_, i) => ({
    css: '',
    defs: '',
    elementos:
      `<g class="alb" style="--d:${ATRASOS[i]}">` +
      `<g class="vinil">${vinil(i, true)}</g>` +
      capaComReflexo(i) +
      '<rect class="sombra" x="-44" y="-44" width="88" height="120" fill="#0e4429"/></g>',
  }))
  const titulos = albuns.map((album, i) => {
    const t = titulo(i, album, 200, 360, true)
    return { ...t, elementos: `<g class="tit" style="--d:${ATRASOS[i]}">${t.elementos}</g>` }
  })
  const indicador =
    '<g transform="translate(350 14)">' +
    ATRASOS.map((d, i) => `<rect class="pt" x="${i * 11}" width="8" height="8" rx="2" style="--d:${d}"/>`).join('') +
    '</g>'
  const fundo =
    '<ellipse class="brilho" cx="200" cy="70" rx="95" ry="55" fill="url(#glow)"/>' +
    '<ellipse cx="200" cy="116" rx="120" ry="9" fill="url(#piso)"/>'
  const corpo = juntar([...discos, ...titulos])
  return { css: CSS_GIRO + CSS_CARROSSEL + corpo.css, defs: corpo.defs, elementos: fundo + indicador + corpo.elementos }
}

/** 1 ou 2 álbuns: parados, centralizados, com o vinil para fora girando. */
function estatico(albuns: Album[]): Parte {
  const posicoes = albuns.length === 1 ? [184] : [105, 265]
  const largura = albuns.length === 1 ? 360 : 150
  const partes = albuns.map((album, i) => {
    const t = titulo(i, album, posicoes[i], largura, false)
    return {
      css: t.css,
      defs: t.defs,
      elementos:
        `<g transform="translate(${posicoes[i]} 68)"><g transform="translate(32 0)">${vinil(i, true)}</g>${capaComReflexo(i)}</g>` +
        t.elementos,
    }
  })
  const corpo = juntar(partes)
  return { css: CSS_GIRO + corpo.css, defs: corpo.defs, elementos: '<ellipse cx="200" cy="116" rx="120" ry="9" fill="url(#piso)"/>' + corpo.elementos }
}

function vazio(mensagem: string): Parte {
  const discos = [110, 200, 290].map((x) => `<g transform="translate(${x} 70)">${vinil(-1, false)}</g>`).join('')
  return {
    css: '',
    defs: '',
    elementos:
      `<g opacity=".35">${discos}</g>` +
      `<text x="200" y="136" text-anchor="middle" font-size="12" fill="#ffffff" fill-opacity=".8" font-family="${FONTE}">${escaparXml(mensagem)}</text>`,
  }
}

export function renderizarVitrine(
  albuns: Album[],
  capas: (string | null)[],
  agora: Date,
  idioma: Idioma = 'en',
): string {
  const t = TEXTOS[idioma]
  const lista = albuns.slice(0, 3)
  const tipo = lista.length === 3 ? 'carrossel' : lista.length > 0 ? 'estatico' : 'vazio'
  const corpo = tipo === 'carrossel' ? carrossel(lista) : tipo === 'estatico' ? estatico(lista) : vazio(t.vitrineVazia)
  const rotulo =
    tipo === 'vazio' ? t.vitrineVazia : t.rotuloVitrine(lista.map((a) => `${a.nome} — ${a.artistas}`).join('; '))
  const cabecalho = `${t.vitrineTitulo} · ${t.meses[agora.getUTCMonth()]}`
  const imagens = lista.map((_, i) => defImagem(i, capas[i] ?? null)).join('')

  return [
    `<svg xmlns="http://www.w3.org/2000/svg" width="400" height="150" viewBox="0 0 400 150" class="${tipo}" role="img" aria-label="${escaparXml(rotulo)}">`,
    `<title>${escaparXml(rotulo)}</title>`,
    `<style>${corpo.css}</style>`,
    `<defs>${DEFS_FIXOS}${imagens}${corpo.defs}</defs>`,
    `<rect width="400" height="150" rx="10" fill="${FUNDO}"/>`,
    `<text x="16" y="22" font-size="10" letter-spacing="1.5" fill="#9be9a8" font-family="${FONTE}">${escaparXml(cabecalho)}</text>`,
    corpo.elementos,
    '</svg>',
  ].join('')
}

import type { Dependencias } from '../gerar-card.js'
import type { Idioma } from '../idioma.js'
import { baixarCapaBase64, lerCredenciais, obterAccessToken } from '../spotify.js'
import { escolherAlbuns } from './escolher-albuns.js'
import { buscarItensPlaylist, lerIdPlaylist } from './playlist.js'
import { renderizarVitrine } from './render-vitrine.js'
import type { Album } from './tipos.js'

export const CABECALHOS_VITRINE: Record<string, string> = {
  'Cache-Control': 'no-cache, no-store, max-age=0',
  'CDN-Cache-Control': 'max-age=3600, stale-while-revalidate=86400',
}

export async function obterAlbuns(deps: Dependencias): Promise<Album[]> {
  const id = lerIdPlaylist(deps.env.SPOTIFY_SHOWCASE_PLAYLIST)
  if (!id) {
    deps.log('SPOTIFY_SHOWCASE_PLAYLIST ausente ou inválida', null)
    return []
  }
  try {
    const token = await obterAccessToken(lerCredenciais(deps.env), deps.fetchFn)
    const dados = await buscarItensPlaylist(token, id, deps.fetchFn)
    return escolherAlbuns(Array.isArray(dados?.items) ? dados.items : [])
  } catch (erro) {
    deps.log('Falha ao ler a playlist da vitrine', erro)
    return []
  }
}

export async function gerarVitrine(deps: Dependencias, idioma: Idioma = 'en'): Promise<string> {
  const albuns = await obterAlbuns(deps)
  try {
    const capas = await Promise.all(
      albuns.map((a) => (a.capaUrl ? baixarCapaBase64(a.capaUrl, deps.fetchFn) : Promise.resolve(null))),
    )
    return renderizarVitrine(albuns, capas, deps.agora(), idioma)
  } catch (erro) {
    deps.log('Falha ao montar a vitrine', erro)
    return renderizarVitrine([], [], deps.agora(), idioma)
  }
}
